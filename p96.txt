#include <stdio.h>
int main() {
int n, a, b, size = 15; 
for (a = size/2; a <= size; a++) {
for (b = 1; b < size-a; b++) {
printf(" ");
}
for (b = 1; b <= a; b++) {
printf("*");
}
for (b = 1; b <= size-a; b++) {
printf(" ");
}
printf(" ");
}
for (a = size; a >= 0; a--) {
for (b = a; b < size; b++) {
printf(" ");
}
printf("*");
printf(" ");
}
return 0;
}
