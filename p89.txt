#include<stdio.h>
#include<math.h>
int main()
{
    int i,n;
    int term,sum=0;
    printf("\n Enter the valuie of n:");
    scanf("%d", &n);
    for(i=1;i<=n;i++)
        {
            if(i % 2 == 0)
            {
                term=pow(i , 2);
                sum=sum += term;
            }
        }
    printf("\n 2sqr+3sqr+...=%d",sum);
    return 0;
}
